cmake_minimum_required(VERSION 3.20)
project(FirstPersonNarrativeLauncher LANGUAGES CXX RC)

configure_file(FirstPersonMod.ini "${CMAKE_CURRENT_BINARY_DIR}/LauncherDefault.ini" COPYONLY)
configure_file(src/LauncherResources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/LauncherResources.rc" @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/LauncherResources.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/LauncherDefault.ini;${CMAKE_CURRENT_SOURCE_DIR}/assets/launcher/logo.png;${CMAKE_CURRENT_SOURCE_DIR}/assets/launcher/FirstPersonNarrativeLauncher.ico;${CMAKE_CURRENT_SOURCE_DIR}/assets/launcher/Inter-SemiBold.ttf;${CMAKE_CURRENT_SOURCE_DIR}/assets/launcher/Inter-Medium.ttf;${CMAKE_CURRENT_SOURCE_DIR}/assets/launcher/Inter-OFL.txt")
add_executable(FirstPersonNarrativeLauncher WIN32 src/Launcher.cpp src/LauncherSettings.cpp "${CMAKE_CURRENT_BINARY_DIR}/LauncherResources.rc")
target_compile_features(FirstPersonNarrativeLauncher PRIVATE cxx_std_20)
target_compile_definitions(FirstPersonNarrativeLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(FirstPersonNarrativeLauncher PRIVATE comctl32 gdiplus ole32 shell32)
set_target_properties(FirstPersonNarrativeLauncher PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    target_compile_options(FirstPersonNarrativeLauncher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(LauncherSettingsTests tests/LauncherSettings.cpp src/LauncherSettings.cpp)
    target_compile_features(LauncherSettingsTests PRIVATE cxx_std_20)
    target_compile_definitions(LauncherSettingsTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(LauncherSettingsTests PRIVATE /utf-8)
    endif()
    add_test(NAME LauncherSettings COMMAND LauncherSettingsTests)
endif()
